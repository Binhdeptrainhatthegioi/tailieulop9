import tkinter as tk
import threading
import random
import os
import time


# ── Trạng thái toàn cục ──────────────────────────────────────
is_running = False
windows    = []
main_root  = None


def close_all():
    """Tắt toàn bộ chương trình."""
    global is_running
    is_running = False
    for w in windows:
        try: w.destroy()
        except: pass
    os._exit(0)


def auto_close_window(win):
    """Đóng một cửa sổ và xóa khỏi danh sách."""
    try:
        if win in windows:
            windows.remove(win)
        win.destroy()
    except: pass


def spawn_window():
    """Tạo một cửa sổ pop-up mới và lên lịch cái tiếp theo."""
    if not is_running:
        return

    win = tk.Toplevel(main_root)
    win.overrideredirect(True)   # Ẩn thanh tiêu đề hệ thống

    w, h = 550, 250
    sw   = win.winfo_screenwidth()
    sh   = win.winfo_screenheight()
    x    = random.randint(-100, sw - 100)
    y    = random.randint(-100, sh - 100)

    win.geometry(f"{w}x{h}+{x}+{y}")
    win.attributes('-topmost', True)
    win.configure(bg='black')

    # Thanh tiêu đề giả màu tối
    title_bar = tk.Frame(win, bg="#222222", relief="flat", bd=0)
    title_bar.pack(expand=False, fill="x", side="top")

    tk.Label(
        title_bar, text="Tsbinh",
        bg="#222222", fg="white",
        font=("Arial", 10)
    ).pack(side="left", padx=10, pady=2)

    # Nội dung chính
    tk.Label(
        win, text="Hacked by Binh",
        font=("Arial", 50, "bold"),
        bg='black', fg='red'
    ).pack(expand=True)

    windows.append(win)

    # Tự đóng sau 2.5 giây
    win.after(2500, lambda: auto_close_window(win))

    # Kích hoạt cửa sổ tiếp theo sau 40 ms
    if is_running:
        main_root.after(40, spawn_window)


def wait_for_stop_key():
    """Luồng nền: nhấn Enter trong terminal để TẮT chương trình."""
    input()   # Chặn đến khi người dùng nhấn Enter
    close_all()


if __name__ == "__main__":
    print("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
    print("  💀  Hacked by Binh — Pop-up Storm")
    print("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
    print("► Nhấn ENTER để BẮT ĐẦU pop-up liên tục...")
    input()   # Chờ người dùng nhấn Enter lần 1

    print("► Đang chạy! Nhấn ENTER lần nữa để DỪNG.")

    is_running = True

    # Luồng lắng nghe phím Enter để dừng
    stop_thread = threading.Thread(target=wait_for_stop_key, daemon=True)
    stop_thread.start()

    # Khởi tạo cửa sổ gốc ẩn
    main_root = tk.Tk()
    main_root.withdraw()

    # Bắt đầu chuỗi pop-up
    spawn_window()

    # Vòng lặp giao diện
    main_root.mainloop()
